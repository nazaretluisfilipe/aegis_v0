create type public.user_role     as enum ('admin','parent','student','teacher');
create type public.lesson_status as enum ('scheduled','done','canceled');
create type public.lead_status   as enum ('new','contacted','negotiating','won','lost');

-- ---------- PERFIS (1 por usuário do Supabase Auth) ----------
create table public.profiles (
  id         uuid primary key references auth.users(id) on delete cascade,
  role       public.user_role not null default 'parent',
  full_name  text not null default '' check (char_length(full_name) <= 120),
  phone      text check (char_length(phone) <= 30),
  active     boolean not null default true,
  created_at timestamptz not null default now()
);

-- O papel vem de app_metadata (só o service role consegue gravar; o usuário NÃO).
create function public.handle_new_user() returns trigger
language plpgsql security definer set search_path = public as $$
begin
  insert into public.profiles (id, role, full_name)
  values (
    new.id,
    coalesce(nullif(new.raw_app_meta_data->>'role',''), 'parent')::public.user_role,
    coalesce(new.raw_user_meta_data->>'full_name','')
  );
  return new;
end $$;
create trigger on_auth_user_created after insert on auth.users
  for each row execute function public.handle_new_user();

-- ---------- FUNÇÕES DE AUTORIZAÇÃO ----------
create function public.is_admin() returns boolean
language sql stable security definer set search_path = public as $$
  select exists (select 1 from profiles where id = auth.uid() and role = 'admin' and active)
$$;

create table public.subjects (
  id   uuid primary key default gen_random_uuid(),
  name text not null unique check (char_length(name) <= 80),
  active boolean not null default true
);

create table public.teachers (
  id          uuid primary key default gen_random_uuid(),
  profile_id  uuid unique references public.profiles(id) on delete set null,
  full_name   text not null check (char_length(full_name) <= 120),
  bio         text check (char_length(bio) <= 2000),
  specialties text[] not null default '{}',
  availability text check (char_length(availability) <= 500),
  photo_path  text,                          -- caminho no bucket público 'site'
  show_on_site boolean not null default false,
  active      boolean not null default true,
  created_at  timestamptz not null default now()
);
create table public.teacher_subjects (
  teacher_id uuid references public.teachers(id) on delete cascade,
  subject_id uuid references public.subjects(id) on delete cascade,
  primary key (teacher_id, subject_id)
);

create table public.students (
  id         uuid primary key default gen_random_uuid(),
  profile_id uuid unique references public.profiles(id) on delete set null, -- login do aluno (opcional)
  full_name  text not null check (char_length(full_name) <= 120),
  birth_date date,
  grade      text check (char_length(grade) <= 80),
  school     text check (char_length(school) <= 120),
  goal       text check (char_length(goal) <= 500),
  active     boolean not null default true,
  created_at timestamptz not null default now()
);
-- Responsável = profile com role 'parent'. Vínculo N:N.
create table public.parent_students (
  parent_id  uuid references public.profiles(id) on delete cascade,
  student_id uuid references public.students(id) on delete cascade,
  primary key (parent_id, student_id)
);

create function public.can_access_student(sid uuid) returns boolean
language sql stable security definer set search_path = public as $$
  select public.is_admin()
      or exists (select 1 from students s where s.id = sid and s.active and s.profile_id = auth.uid())
      or exists (select 1 from parent_students ps join profiles p on p.id = ps.parent_id
                 where ps.student_id = sid and ps.parent_id = auth.uid() and p.active)
$$;

create table public.lessons (
  id           uuid primary key default gen_random_uuid(),
  student_id   uuid not null references public.students(id) on delete cascade,
  teacher_id   uuid references public.teachers(id) on delete set null,
  subject_id   uuid references public.subjects(id) on delete set null,
  starts_at    timestamptz not null,
  duration_min int not null default 50 check (duration_min between 15 and 240),
  meet_url     text check (meet_url is null or meet_url ~ '^https://meet\.google\.com/[A-Za-z0-9?=&_/.-]+$'),
  status       public.lesson_status not null default 'scheduled',
  created_at   timestamptz not null default now()
);
create index on public.lessons (student_id, starts_at desc);

-- Feedback da aula, visível à família só se publicado.

);

create table public.materials (
  id           uuid primary key default gen_random_uuid(),
  student_id   uuid not null references public.students(id) on delete cascade,
  lesson_id    uuid references public.lessons(id) on delete set null,
  title        text not null check (char_length(title) <= 160),
  description  text check (char_length(description) <= 1000),
  storage_path text not null unique,     -- no bucket PRIVADO 'materials'
  created_at   timestamptz not null default now()

create table public.reports (
  id          uuid primary key default gen_random_uuid(),
  student_id  uuid not null references public.students(id) on delete cascade,
  lesson_id   uuid references public.lessons(id) on delete set null,
  subject_id  uuid references public.subjects(id) on delete set null,

  content_worked text, performance text, difficulties text, positives text,

  published   boolean not null default false,
  created_at  timestamptz not null default now(),
  updated_at  timestamptz not null default now()
);

-- ---------- COMERCIAL / CMS ----------
create table public.leads (
  id uuid primary key default gen_random_uuid(),
  status public.lead_status not null default 'new',
  guardian_name text not null check (char_length(guardian_name) between 3 and 120),
  whatsapp text not null check (char_length(whatsapp) between 10 and 20),
  email text not null check (char_length(email) <= 160 and email ~ '^\S+@\S+\.\S+$'),
  student_name text not null check (char_length(student_name) between 1 and 120),
  age text check (char_length(age) <= 10),
  grade text check (char_length(grade) <= 80),
  subjects text[] not null default '{}' check (cardinality(subjects) <= 10),
  goal text check (char_length(goal) <= 200),
  difficulties text check (char_length(difficulties) <= 2000),
  availability text check (char_length(availability) <= 500),
  frequency text check (char_length(frequency) <= 80),
  notes text check (char_length(notes) <= 2000),
  consent boolean not null check (consent),            -- sem consentimento, não grava
  consent_at timestamptz not null default now(),

-- Admin altera papel/ativo de perfis (única via)
create function public.admin_set_profile(pid uuid, new_role public.user_role, is_active boolean) returns void
language plpgsql security definer set search_path = public as $$
begin
  if not public.is_admin() then raise exception 'forbidden'; end if;
  update profiles set role = new_role, active = is_active where id = pid;
end $$;
revoke all on function public.admin_set_profile from public, anon;
grant execute on function public.admin_set_profile to authenticated;

create function public.touch_updated_at() returns trigger language plpgsql as $$ begin new.updated_at = now(); return new; end $$;
create trigger reports_touch before update on public.reports for each row execute function public.touch_updated_at();
create trigger content_touch before update on public.site_content for each row execute function public.touch_updated_at();
